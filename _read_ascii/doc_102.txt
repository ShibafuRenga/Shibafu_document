===FILE: SymbolBlur.shader===
Shader "UI/SymbolBlur"
{
    Properties
    {
        [PerRendererData] _MainTex ("Texture", 2D) = "white" {}
        _Color ("Tint", Color) = (1, 1, 1, 1)

        [HideInInspector] _StencilComp ("Stencil Comparison", Float) = 8
        [HideInInspector] _Stencil ("Stencil ID", Float) = 0
        [HideInInspector] _StencilOp ("Stencil Operation", Float) = 0
        [HideInInspector] _StencilWriteMask ("Stencil Write Mask", Float) = 255
        [HideInInspector] _StencilReadMask ("Stencil Read Mask", Float) = 255
        [HideInInspector] _ColorMask ("Color Mask", Float) = 15
        [Toggle(UNITY_UI_ALPHACLIP)] _UseUIAlphaClip ("Use Alpha Clip", Float) = 0
    }

    SubShader
    {
        Tags
        {
            "Queue" = "Transparent"
            "IgnoreProjector" = "True"
            "RenderType" = "Transparent"
            "PreviewType" = "Plane"
            "CanUseSpriteAtlas" = "True"
        }

        Stencil
        {
            Ref [_Stencil]
            Comp [_StencilComp]
            Pass [_StencilOp]
            ReadMask [_StencilReadMask]
            WriteMask [_StencilWriteMask]
        }

        Cull Off
        Lighting Off
        ZWrite Off
        ZTest [unity_GUIZTestMode]
        Blend SrcAlpha OneMinusSrcAlpha
        ColorMask [_ColorMask]

        Pass
        {
            Name "SymbolBlur"

            CGPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #pragma target 2.0

            #include "UnityCG.cginc"
            #include "UnityUI.cginc"

            #pragma multi_compile_local _ UNITY_UI_CLIP_RECT
            #pragma multi_compile_local _ UNITY_UI_ALPHACLIP

            struct appdata_t
            {
                float4 vertex : POSITION;
                fixed4 color : COLOR;
                float2 uv : TEXCOORD0;
                float4 uvBounds : TEXCOORD1;
                float4 blurParameters : TEXCOORD2;
                UNITY_VERTEX_INPUT_INSTANCE_ID
            };

            struct v2f
            {
                float4 vertex : SV_POSITION;
                fixed4 color : COLOR;
                float2 uv : TEXCOORD0;
                float4 uvBounds : TEXCOORD1;
                float4 blurParameters : TEXCOORD2;
                float4 worldPosition : TEXCOORD3;
                UNITY_VERTEX_OUTPUT_STEREO
            };

            sampler2D _MainTex;
            float4 _MainTex_TexelSize;
            fixed4 _Color;
            float4 _ClipRect;

            v2f vert(appdata_t v)
            {
                v2f o;
                UNITY_SETUP_INSTANCE_ID(v);
                UNITY_INITIALIZE_VERTEX_OUTPUT_STEREO(o);

                o.worldPosition = v.vertex;
                o.vertex = UnityObjectToClipPos(v.vertex);
                o.uv = v.uv;
                o.uvBounds = v.uvBounds;
                o.blurParameters = v.blurParameters;
                o.color = v.color * _Color;
                return o;
            }

            fixed4 SampleSprite(float2 uv, float4 uvBounds, float4 safeUvBounds)
            {
                // 扩展区域的 UV 会落在原 Sprite 外。先限制真实纹理采样位置，
                // 再用原始 UV 边界清零，防止 SpriteAtlas 中相邻图案串色。
                float2 safeUv = clamp(uv, safeUvBounds.xy, safeUvBounds.zw);

                float inside = step(uvBounds.x, uv.x) * step(uvBounds.y, uv.y) *
                               step(uv.x, uvBounds.z) * step(uv.y, uvBounds.w);
                return tex2D(_MainTex, safeUv) * inside;
            }

            float InterleavedGradientNoise(float2 pixelPosition)
            {
                // 稳定的屏幕空间抖动，让相邻像素使用不同的采样相位。
                return frac(52.9829189 * frac(dot(floor(pixelPosition), float2(0.06711056, 0.00583715))));
            }

            fixed4 frag(v2f i) : SV_Target
            {
                // 高速滚轴需要的是沿运动方向的连续拖影，而不是普通的二维模糊。
                // 17 个分层采样点在控制开销的同时形成连续速度线。
                const int sampleCount = 17;
                float verticalBlur = max(0.0, i.blurParameters.x);
                float blurStrength = saturate(i.blurParameters.y);
                float antiBanding = saturate(i.blurParameters.z);
                float motion = _MainTex_TexelSize.y * verticalBlur;

                // 没挂 SymbolBlur 组件时 TEXCOORD1 为 0，退回整张纹理范围，
                // 保证材质仍能正常显示，方便在编辑器中预览。
                float hasUvBounds = step(0.0000001,
                    (i.uvBounds.z - i.uvBounds.x) * (i.uvBounds.w - i.uvBounds.y));
                float4 uvBounds = lerp(float4(0.0, 0.0, 1.0, 1.0), i.uvBounds, hasUvBounds); 这个uvBonunds的判断不应该带到运行打包之后,打包后肯定全是有uvBounds的
                float2 halfTexel = _MainTex_TexelSize.xy * 0.5;
                float4 safeUvBounds = uvBounds + float4(halfTexel, -halfTexel);

                // 在预乘 Alpha 空间累加，透明 PNG 的边缘不会被透明区 RGB 污染。
                float4 blurred = 0;
                float totalWeight = 0;
                float noise = InterleavedGradientNoise(i.vertex.xy) - 0.5;
                [unroll]
                for (int tap = 0; tap < sampleCount; tap++)
                {
                    // 每个采样位于自己的区间内，抖动只改变区间内位置，不产生采样空洞。
                    // 数值 10 表示以原图为中心，向两侧各延伸 10 像素。
                    float tapPosition = tap + 0.5 + noise * antiBanding;
                    float t = (tapPosition / sampleCount) * 2.0 - 1.0;

                    // 宽缓权重保留运动感，同时削弱最外侧的重复轮廓。
                    float weight = 1.0 - abs(t) * 0.25;
                    fixed4 sampleColor = SampleSprite(
                        i.uv + float2(0.0, motion * t),
                        uvBounds, // uvBonuds是实际在图集里面，uv的rect区域
                        safeUvBounds); // 图集里面的uv rect区域上下各加原图大小的一半？意义不明
                    blurred.rgb += sampleColor.rgb * sampleColor.a * weight;
                    blurred.a += sampleColor.a * weight;
                    totalWeight += weight;
                }
                blurred /= totalWeight;
                blurred.rgb /= max(blurred.a, 0.0001);

                fixed4 source = SampleSprite(i.uv, uvBounds, safeUvBounds);
                fixed4 color = lerp(source, blurred, blurStrength);
                color *= i.color;

                #ifdef UNITY_UI_CLIP_RECT
                color.a *= UnityGet2DClipping(i.worldPosition.xy, _ClipRect);
                #endif

                #ifdef UNITY_UI_ALPHACLIP
                clip(color.a - 0.001);
                #endif

                return color;
            }
            ENDCG
        }
    }
}
